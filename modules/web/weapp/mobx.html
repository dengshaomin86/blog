import { observable, action } from "mobx-miniprogram";

export const userStore = observable({
  name: "",
  setName: action(function (name: string) {
    this.name = name;
  }),

import { observable, action } from "mobx-miniprogram";

  count: 0,
  inc: action(function () {
    this.count++;
  }),

import { counterStore } from "./counter";

export const rootStore = {
  user: userStore,
  counter: counterStore,

import { counterStore } from "@/stores/counter";

Component({

  storeBindings: {
    fields: {
      name: () => userStore.name,
      count: () => counterStore.count,
    },
    actions: {
      setName: () => userStore.setName,
      inc: () => counterStore.inc,
    },
  },
});

// Component rootStore
import { rootStore } from "@/stores/root";

Component({
  behaviors: [storeBindingsBehavior],

  storeBindings: {
    fields: {
      name: () => rootStore.user.name,
      count: () => rootStore.counter.count,
    },
    actions: {
      setName: () => rootStore.user.setName,
      inc: () => rootStore.counter.inc,
    },
  },
});

// Page 必须手动绑定
// Page 不支持 behaviors 自动注入生命周期
// 所以 storeBindingsBehavior 不会自动初始化
import { createStoreBindings } from "mobx-miniprogram-bindings";
import { counterStore } from "@/stores/counter";

Page({
  data: {
    count: 0,
  },

  onLoad() {
    this.storeBindings = createStoreBindings(this, {
      fields: {
        count: () => counterStore.count,
      },
      actions: {
        inc: () => counterStore.inc,
      },
    });
  },

  onUnload() {
    this.storeBindings.destroyStoreBindings();
  },
});

// 封装
export function withStoreBindings(config) {
  const { storeBindings, ...rest } = config;

  return Page({
    ...rest,

    onLoad(options) {
      this.storeBindings = createStoreBindings(this, storeBindings);
      rest.onLoad?.call(this, options);
    },

    onUnload() {
      this.storeBindings?.destroyStoreBindings();
      rest.onUnload?.call(this);

// 实现一：useStore（Page 用）
// utils/useStore.ts
import { createStoreBindings } from "mobx-miniprogram-bindings";

type StoreMap = Record<string, any>;

export function useStore(ctx: any, map: StoreMap) {
  const fields: Record<string, any> = {};
  const actions: Record<string, any> = {};

  Object.keys(map).forEach((key) => {
    const val = map[key];

    if (typeof val === "function") {
      // 判断是字段还是action
      // 简单策略：执行一次看返回值
      // 隐患: 有些 field 可能也会 throw, action 也可能返回值
      try {
        const result = val();
        fields[key] = val;
      } catch {
        actions[key] = val;
      }
    }
  });

  const bindings = createStoreBindings(ctx, {
    fields,
    actions,
  });

  // 自动挂载销毁
  const oldUnload = ctx.onUnload;
  ctx.onUnload = function () {
    bindings.destroyStoreBindings();
    oldUnload && oldUnload.call(this);
  };

  // 挂到实例（方便调试）
  ctx.__storeBindings = bindings;
}

// 实现二：withStore（Component 用）
// utils/useStore.ts
import { storeBindingsBehavior } from "mobx-miniprogram-bindings";

export function withStore(options: any) {
  const { store = {}, ...rest } = options;

  const fields: Record<string, any> = {};
  const actions: Record<string, any> = {};

  Object.keys(store).forEach((key) => {
    const val = store[key];

    if (typeof val === "function") {
      try {
        val();
        fields[key] = val;
      } catch {
        actions[key] = val;
      }
    }
  });

  return {
    behaviors: [storeBindingsBehavior],

    storeBindings: {
      fields,
      actions,
    },

    ...rest,
  };
}

进阶玩法3 ​

ts
import { createStoreBindings, storeBindingsBehavior } from 'mobx-miniprogram-bindings';

export function useStore(ctx: any, options: any) {
  const bindings = createStoreBindings(ctx, {
    store: options.store,
    fields: options.fields || {},
    actions: options.actions || {},
  });

  const oldUnload = ctx.onUnload || function () { };

  // 如果 useStore 被调用多次 → 会覆盖，生命周期顺序可能错乱
  ctx.onUnload = function (...args: any[]) {
    bindings.destroyStoreBindings();
    return oldUnload.apply(this, args);
  };
}

export function withStore(options: any) {
  const userBehaviors = options.behaviors || []
  return {
    behaviors: [
      storeBindingsBehavior,
      ...userBehaviors,
    ],
    storeBindings: {
      store: options.store,
      fields: options.fields || {},
      actions: options.actions || {},
    },
    ...options,
  };
}

// Page
onLoad() {
  useStore(this, {
    store: counterStore,
    fields: {
      count: 'count',
    },
    actions: {
      inc: 'inc',
    },
  });
}

// Page 使用多个 store
useStore(this, {

  },
  actions: {
    inc: 'counter.inc',

});

custom-tab-bar ​

不是普通组件 是由小程序框架单独实例化和管理 不会走完整的页面/组件更新链路

👉 结果就是：

MobX 能更新数据，但 UI 不一定刷新

tabBar 组件实例是“单例”

👉 意味着：

不会像页面一样频繁销毁/创建 storeBindings 生命周期可能不触发

用 autorun + setData

ts
// custom-tab-bar/index.ts
import { autorun } from "mobx-miniprogram";
import { counterStore } from "@/stores/counter";

Component({
  data: {
    count: 0,
  },

  lifetimes: {
    attached() {
      this.disposer = autorun(() => {
        this.setData({
          count: counterStore.count,
        });
      });
    },

    detached() {
      this.disposer?.();
    },
  },
});

封装：

ts
type Disposer = () => void;

export function bindAutorun(ctx: any, fn: () => void) {
  const disposer: Disposer = autorun(fn);

  const oldDetached = ctx.detached || function () {};

  ctx.detached = function (...args: any[]) {
    disposer();
    return oldDetached.apply(this, args);
  };
}

// custom-tab-bar/index.ts
lifetimes: {
  attached() {
    bindAutorun(this, () => {
      this.setData({
        count: counterStore.count
      })
    })
  }
},
