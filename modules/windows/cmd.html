On this page

cmd ​

  • ⚠️ ‼️文件编码格式不要选择utf8，否则有中文会报错。选 ANSI/GBK⚠️
bat
@echo off
git pull
echo 按任意键退出...
echo.
@REM pause > nul（隐藏 "Press any key to continue..." 默认提示）
pause > nul

说明 ​

  • @echo off = "让脚本运行时只显示结果，不要暴露代码本身"，是批处理脚本的标准开头配置。
  • @ 符号的作用：不显示当前行命令
  • @REM Pause 只是 注释（REM = remark），不会执行任何操作。
  • pause > nul（隐藏 "Press any key to continue..." 默认提示）
  • echo. 相当于换行符
  • %cd% 是在脚本解析时就展开的变量；!cd! 是运行时动态获取变量。

cd 在不同盘符间切换 /d ​

bat
cd /d xxx

timeout（等待几秒后自动退出） ​

bat
@echo off
git pull
timeout /t 5 /nobreak > nul

choice（让用户选择是否退出） ​

bat
@echo off
git pull
choice /c YN /m "是否退出？[Y/N]"
if %errorlevel% equ 1 exit

文件选择 ​

使用 PowerShell 调用文件选择对话框 ​

  • 如需选择文件夹而非文件，将 OpenFileDialog 改为 FolderBrowserDialog（PowerShell）
bat
@echo off
for /f "delims=" %%i in ('powershell -command "[System.Reflection.Assembly]::LoadWithPartialName('System.Windows.Forms') | Out-Null; $dlg = New-Object System.Windows.Forms.OpenFileDialog; $dlg.Title = '选择文件'; $dlg.Filter = '所有文件 (*.*)|*.*'; if ($dlg.ShowDialog() -eq 'OK') { $dlg.FileName }"') do (
    set "filepath=%%i"
)
echo 你选择的文件路径是: "%filepath%"
pause

使用 mshta 调用 HTML 对话框 ​

bat
@echo off
for /f "delims=" %%i in ('mshta "javascript:var f=new ActiveXObject('Scripting.FileSystemObject').GetFile(inputbox('输入文件路径:')).Path;close(new ActiveXObject('Scripting.FileSystemObject').CreateTextFile('%~dp0tmp.txt',true).Write(f));"^<nul & type "%~dp0tmp.txt" & del "%~dp0tmp.txt"') do (
    set "filepath=%%i"
)
echo 你选择的路径是: "%filepath%"
pause

纯 CMD 手动输入路径 ​

bat
@echo off
set /p filepath=请拖入文件或输入完整路径: 
echo 你选择的路径是: "%filepath%"
pause

call ​

bat
:: 另一个 .bat 或 .cmd 文件需要用 call,不加 call 会中断当前脚本，直接跳过去执行，之后不返回
call other-script.bat

:: .exe 可执行程序（如 git, node, xcopy, npm 等）不需要 call，自动阻塞执行，执行完才继续
git add .
git commit -m "docs: 自动更新 %date% %time%"
git push

示例 ​

bat
@echo off
setlocal enabledelayedexpansion
title 打包文档
color 0B

set DEFAULT_OUTPUT_DIR=d:\pages\dist
set OUTPUT_DIR=%DEFAULT_OUTPUT_DIR%
set DIST_DIR=.vitepress\dist

call npm run docs:build
echo 构建完成！

:menu
echo [1] 复制输出文件到 pages 目录
echo [2] 退出
choice /c 12 /n /m "请选择:"
if %errorlevel% equ 1 goto copy
if %errorlevel% equ 2 exit

:copy
echo 请选择复制目录方式：
echo [1] 使用默认目录：%DEFAULT_OUTPUT_DIR%
echo [2] 手动选择目录
echo [3] 退出
choice /c 123 /n /m "请选择:"
if %errorlevel% equ 2 (
  set "OUTPUT_DIR_SELECTED="
  for /f "delims=" %%i in (
    'powershell -nologo -command "Add-Type -AssemblyName System.Windows.Forms; $f=New-Object Windows.Forms.FolderBrowserDialog; $f.Description='请选择输出目录'; $f.SelectedPath='!DEFAULT_OUTPUT_DIR!'; if($f.ShowDialog() -eq 'OK'){ $f.SelectedPath }"'
  ) do (
    set "OUTPUT_DIR_SELECTED=%%i"
  )

  :: 如果为空，说明用户取消了选择
  if not defined OUTPUT_DIR_SELECTED (
    echo 已取消选择，返回菜单
    goto copy
  )

  set "OUTPUT_DIR=!OUTPUT_DIR_SELECTED!"
  echo 输出路径：!OUTPUT_DIR!
  pause
)
if %errorlevel% equ 3 exit

echo ### 开始复制 ###

xcopy /E /Y "!DIST_DIR!" "!OUTPUT_DIR!\" || (
  echo 复制失败
  pause
  exit /b 1
)
echo ### 复制完成 ###
goto chagedir


:chagedir
echo [1] 切换到输出目录: !OUTPUT_DIR!
echo [2] 退出
choice /c 12 /n /m "请选择:"
if %errorlevel% equ 1 (
    cd /d "!OUTPUT_DIR!" || (
        echo 切换目录失败：!OUTPUT_DIR!
        pause
        exit /b 1
    )
    echo 当前目录：!cd!
    goto push
) else (
    exit
)

:push
echo [1] push
echo [2] 退出
choice /c 12 /n /m "请选择:"
if %errorlevel% equ 1 (
    echo 正在提交变更
    git add .
    git commit -m "docs: 自动更新 %date% %time%"
    git push
) else (
    exit
)
echo 按任意键退出
pause > nul
